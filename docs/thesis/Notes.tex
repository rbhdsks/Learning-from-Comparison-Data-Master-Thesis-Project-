\chapter{Introduction}
\section{Motivation}
Let us imagine that someone has to travel from IIT Madras to the airport.
The person has four realistic choices: car, bike, taxi, and bus. Based on
cost, comfort, and speed, the person may have a natural preference ranking
among these alternatives. For example, the ranking may be
\begin{equation}
    \text{Car} \succ \text{Taxi} \succ \text{Bike} \succ \text{Bus}.
\end{equation}
Now suppose that a fifth alternative is added. This alternative is walking,
which may take almost three hours and is therefore unlikely to be seriously
considered.
Common sense suggests that adding walking should not change the person's
relative preference between car, taxi, bike, and bus. Walking is an
irrelevant alternative in this situation. Therefore, the relative preference
between the original alternatives should remain unchanged.
This idea is called the \emph{Independence of Irrelevant Alternatives}
property. It is one of the central properties behind classical models of
choice.
However, human preferences may not always behave exactly as assumed by
classical multinomial logit models. Experimental evidence indicates that
adding or removing alternatives can change how attractive the existing
alternatives appear. Such behaviour is described as a violation of the
Independence of Irrelevant Alternatives property
\citep{seshadri2019}.

\chapter{Background and Choice Models}
\section{Choice Data and Choice Sets}
Let \(\X\) denote the universal set of alternatives. A choice set is a
subset of \(\X\) containing the alternatives available to a decision-maker
at a particular time.
For the commute-mode example, the universal set is
\begin{equation}
    \X=
    \{
    \text{Car},
    \text{Taxi},
    \text{Bike},
    \text{Bus},
    \text{Walking}
    \}.
\end{equation}
The walking alternative remains a member of the universal set even when it
does not appear in a particular choice set. All the smaller choice sets are
constructed from the same fixed universal set.
Let \(A,B\subseteq\X\) be two choice sets. Let \(x\) and \(y\) be
alternatives belonging to the intersection \(A\cap B\).
\section{Independence of Irrelevant Alternatives}
For a universal set \(\X\) and choice sets \(A,B\subseteq\X\), the
Independence of Irrelevant Alternatives property for two alternatives
\(x\) and \(y\) can be written as
\begin{equation}
    \frac{P(x\mid A)}{P(y\mid A)}
    =
    \frac{P(x\mid B)}{P(y\mid B)},
    \qquad
    x,y\in A\cap B.
    \label{eq:iia}
\end{equation}
The equation states that if \(x\) and \(y\) are present in both choice sets,
then their relative odds should not change merely because other alternatives
have been added or removed.
\subsection{Worked Example}
\begin{table}[htbp]
    \centering
    \caption{Alternatives and their assigned weights}
    \label{tab:alternative-weights}
    \begin{tabular}{lll}
        \toprule
        Item & Symbol & Weight \(w\)\\
        \midrule
        Car     & \(C\)  & \(0.40\)\\
        Taxi    & \(T\)  & \(0.25\)\\
        Bike    & \(Bi\) & \(0.15\)\\
        Bus     & \(Bu\) & \(0.12\)\\
        Walking & \(W\)  & \(0.08\)\\
        \midrule
        Total   &        & \(1.00\)\\
        \bottomrule
    \end{tabular}
\end{table}
Consider the following two choice sets:
\begin{align}
    A &= \{C,T,Bi\},
    &
    \sum_{x\in A}w_x &= 0.80,
    \\
    B &= \{C,T,W\},
    &
    \sum_{x\in B}w_x &= 0.73.
\end{align}
We want to verify whether
\begin{equation}
    \frac{P(C\mid A)}{P(T\mid A)}
    =
    \frac{P(C\mid B)}{P(T\mid B)}.
\end{equation}
For choice set \(A\),
\begin{align}
    P(C\mid A)
    &=
    \frac{w_C}{\sum_{x\in A}w_x}
    =
    \frac{0.40}{0.80}
    =
    0.50,
    \\
    P(T\mid A)
    &=
    \frac{w_T}{\sum_{x\in A}w_x}
    =
    \frac{0.25}{0.80}
    =
    0.3125.
\end{align}
Therefore,
\begin{equation}
    \frac{P(C\mid A)}{P(T\mid A)}
    =
    \frac{0.40/0.80}{0.25/0.80}
    =
    \frac{0.40}{0.25}
    =
    1.6.
\end{equation}
For choice set \(B\),
\begin{align}
    P(C\mid B)
    &=
    \frac{w_C}{\sum_{x\in B}w_x}
    =
    \frac{0.40}{0.73},
    \\
    P(T\mid B)
    &=
    \frac{w_T}{\sum_{x\in B}w_x}
    =
    \frac{0.25}{0.73}.
\end{align}
Therefore,
\begin{equation}
    \frac{P(C\mid B)}{P(T\mid B)}
    =
    \frac{0.40/0.73}{0.25/0.73}
    =
    \frac{0.40}{0.25}
    =
    1.6.
\end{equation}
Thus,
\begin{equation}
    \frac{P(C\mid A)}{P(T\mid A)}
    =
    \frac{P(C\mid B)}{P(T\mid B)}.
\end{equation}
Hence, the Independence of Irrelevant Alternatives property holds in this
worked example.
\section{Multinomial Logit Model}
The classical multinomial logit model assigns a score \(v(x)\) to every
alternative \(x\). The probability of selecting \(x\) from choice set \(C\)
is
\begin{equation}
    P(x\mid C)
    =
    \frac{\exp(v(x))}
    {\displaystyle\sum_{y\in C}\exp(v(y))}.
    \label{eq:mnl}
\end{equation}
The scores are shift-invariant. A common identification constraint is
\begin{equation}
    \sum_{x\in\X}v(x)=0.
    \label{eq:mnl-identification}
\end{equation}
\subsection{Shift Invariance and the Zero-Sum Constraint}
Consider two alternatives, car and taxi, with scores
\begin{align}
    v(C)&=2.5,\\
    v(T)&=-0.9.
\end{align}
Their exponentiated scores are
\begin{align}
    \exp(v(C))
    &=
    \exp(2.5)
    \approx
    12.182,
    \\
    \exp(v(T))
    &=
    \exp(-0.9)
    \approx
    0.407.
\end{align}
Therefore, the probability of choosing car from the set
\(\{C,T\}\) is
\begin{align}
    P(C\mid\{C,T\})
    &=
    \frac{\exp(2.5)}
    {\exp(2.5)+\exp(-0.9)}
    \\
    &=
    \frac{12.182}{12.182+0.407}
    \\
    &\approx
    0.968.
\end{align}
However,
\begin{equation}
    v(C)+v(T)=2.5-0.9=1.6\neq 0.
\end{equation}
Suppose the same constant \(k\) is added to both scores. To satisfy the
zero-sum constraint, choose \(k\) such that
\begin{align}
    (2.5+k)+(-0.9+k)&=0,\\
    1.6+2k&=0,\\
    k&=-0.8.
\end{align}
After applying this shift,
\begin{align}
    v(C)&=2.5-0.8=1.7,\\
    v(T)&=-0.9-0.8=-1.7.
\end{align}
The probability obtained using these centred scores is
\begin{align}
    P(C\mid\{C,T\})
    &=
    \frac{\exp(1.7)}
    {\exp(1.7)+\exp(-1.7)}
    \\
    &=
    \frac{5.474}{5.474+0.183}
    \\
    &\approx
    0.968.
\end{align}
Thus, adding the same constant to every score does not change the choice
probabilities. The zero-sum constraint selects one representative from all
score vectors that produce the same probabilities.

\chapter{From Multinomial Logit to Context-Dependent Choice Models}
\section{Universal Logit Representation}
In a universal logit model, the utility of an alternative can depend on the
choice set in which it appears. Instead of assigning only one fixed utility to
an alternative, the model allows it to have different utilities in different
contexts.
The probability of selecting $x$ from $C$ is
\begin{equation}
    P(x\mid C)
    =
    \frac{\exp(u(x\mid C))}
    {\displaystyle\sum_{y\in C}\exp(u(y\mid C))}.
    \label{eq:context-logit}
\end{equation}
An identification constraint can be imposed separately for each choice set:
\begin{equation}
    \sum_{y\in C}u(y\mid C)=0,
    \qquad
    \text{for every } C\subseteq\X.
\end{equation}
\section{Decomposition of Context-Dependent Utility}
\subsection{Statement of Lemma 1}
\begin{lemma}[Decomposition of context-dependent utility]
For a universal logit model, the context-dependent utility $u(x\mid C)$ can
be written as a sum of terms indexed by subsets $B$ of
$C\setminus\{x\}$ \citep{seshadri2019}:
\begin{equation}
    u(x\mid C)
    =
    \sum_{B\subseteq C\setminus\{x\}}v(x\mid B).
    \label{eq:utility-decomposition}
\end{equation}
Equivalently,
\begin{equation}
\begin{split}
    u(x\mid C)
    &=
    v(x)
    +
    \sum_{y\in C\setminus\{x\}}v(x\mid\{y\}) \\
    &\quad+
    \sum_{\{y,z\}\subseteq C\setminus\{x\}}
    v(x\mid\{y,z\})
    +\cdots.
\end{split}
\end{equation}
The parameters satisfy
\begin{equation}
    \sum_{x\notin B}v(x\mid B)=0
    \qquad
    \text{for every } B\subseteq\X.
    \label{eq:decomposition-constraint}
\end{equation}
\end{lemma}
\subsection{Interpretation of the Different Orders}
The term $v(x)$ represents the intrinsic preference associated with
alternative $x$. It does not depend on the other alternatives present in the
choice set.
The term $v(x\mid\{z\})$ represents the contextual contribution of
alternative $z$ to the utility of alternative $x$. Similarly, a term such as
$v(x\mid\{y,z\})$ represents a higher-order interaction involving the joint
presence of alternatives $y$ and $z$.
\subsection{Worked Example}
Consider the choice set
\begin{equation}
    A=\{C,T,Bi\}.
\end{equation}
The subsets of $A$ are
\begin{equation}
\begin{split}
    \emptyset,\;
    \{C\},\;
    \{T\},\;
    \{Bi\},\;
    \{C,T\},\;
    \{T,Bi\},\;
    \{Bi,C\},\;
    \{C,T,Bi\}.
\end{split}
\end{equation}
The utility expansions are
\begin{align}
    u(C\mid A)
    &=v(C)+v(C\mid\{T\})+v(C\mid\{Bi\})+v(C\mid\{T,Bi\}),\\
    u(T\mid A)
    &=v(T)+v(T\mid\{C\})+v(T\mid\{Bi\})+v(T\mid\{C,Bi\}),\\
    u(Bi\mid A)
    &=v(Bi)+v(Bi\mid\{C\})+v(Bi\mid\{T\})+v(Bi\mid\{C,T\}).
\end{align}
The corresponding constraints are
\begin{align}
    v(C)+v(T)+v(Bi)&=0,\label{eq:constraint-empty}\\
    v(T\mid\{C\})+v(Bi\mid\{C\})&=0,\\
    v(C\mid\{T\})+v(Bi\mid\{T\})&=0,\\
    v(C\mid\{Bi\})+v(T\mid\{Bi\})&=0,\\
    v(C\mid\{T,Bi\})&=0,\\
    v(T\mid\{C,Bi\})&=0,\\
    v(Bi\mid\{C,T\})&=0.
\end{align}
The final three constraints force the highest-order terms to zero. Therefore,
the utilities reduce to
\begin{align}
    u(C\mid A)
    &=v(C)+v(C\mid\{T\})+v(C\mid\{Bi\}),\\
    u(T\mid A)
    &=v(T)+v(T\mid\{C\})+v(T\mid\{Bi\}),\\
    u(Bi\mid A)
    &=v(Bi)+v(Bi\mid\{C\})+v(Bi\mid\{T\}).
\end{align}
There are nine remaining parameters and four independent constraints. Hence,
the number of free parameters is
\begin{equation}
    9-4=5.
    \label{eq:three-alternative-free-parameters}
\end{equation}
\section{Hierarchy of Choice Models}
The decomposition gives a hierarchy of choice models. A lower-order model
retains simpler contextual interactions, while a higher-order model allows the
utility of an alternative to depend on larger subsets of the choice set.
Increasing the order makes the model more flexible, but also increases the
number of parameters that must be estimated.
\subsection{The Model Classes $\mathcal{M}_p$}
Let $\mathcal{M}_p$ denote the model obtained by retaining utility terms up to
order $p$ and setting all higher-order terms equal to zero. This produces the
nested hierarchy
\begin{equation}
    \mathcal{M}_1
    \subseteq
    \mathcal{M}_2
    \subseteq
    \cdots
    \subseteq
    \mathcal{M}_{n-1}.
\end{equation}
The first-order model retains only $v(x)$. The second-order model additionally
retains terms of the form $v(x\mid\{z\})$. Higher-order models include terms
involving larger subsets of alternatives.
\subsection{Number of Free Parameters}
For the second-order model, there are $n$ intrinsic parameters of the form
$v(x)$ and $n(n-1)$ contextual parameters of the form
$v(x\mid\{z\})$, where $x\neq z$. Thus, before applying the constraints,
the number of parameters is
\begin{equation}
    n+n(n-1)=n^2.
\end{equation}
The intrinsic parameters satisfy one constraint:
\begin{equation}
    \sum_{x\in\X}v(x)=0.
\end{equation}
For every $z\in\X$, the contextual parameters satisfy
\begin{equation}
    \sum_{x\in\X\setminus\{z\}}v(x\mid\{z\})=0.
\end{equation}
There are $n$ contextual constraints. Therefore, the total number of
constraints is $n+1$, and the number of free parameters is
\begin{equation}
    n^2-(n+1)=n^2-n-1.
    \label{eq:second-order-parameter-count}
\end{equation}
\subsection{MNL as the First-Order Model}
The multinomial logit model is the first-order member of this hierarchy. In
$\mathcal{M}_1$, all contextual terms are set to zero, leaving
\begin{equation}
    u(x\mid C)=v(x).
\end{equation}
Substituting this expression into the universal logit representation gives
\begin{equation}
    P(x\mid C)
    =
    \frac{\exp(v(x))}
    {\displaystyle\sum_{y\in C}\exp(v(y))},
\end{equation}
which is the multinomial logit model.
\section{Context-Dependent Random Utility Model}
The first-order model does not allow the utility of an alternative to change
when other alternatives are added to or removed from the choice set. To
represent such context effects without retaining every higher-order
interaction, we consider the second-order model $\mathcal{M}_2$.
\subsection{Second-Order Truncation}
Starting from the universal logit decomposition,
\begin{equation}
\begin{split}
    u(x\mid C)
    &=v(x)
    +\sum_{z\in C\setminus\{x\}}v(x\mid\{z\})\\
    &\quad+
    \sum_{\{y,z\}\subseteq C\setminus\{x\}}v(x\mid\{y,z\})
    +\cdots,
\end{split}
\end{equation}
the second-order model sets all terms involving two or more contextual
alternatives equal to zero. Hence,
\begin{equation}
    u(x\mid C)
    =
    v(x)+\sum_{z\in C\setminus\{x\}}v(x\mid\{z\}).
    \label{eq:second-order-truncation}
\end{equation}
The corresponding choice probability is
\begin{equation}
    P(x\mid C)
    =
    \frac{
        \exp\left(
            v(x)+\displaystyle\sum_{z\in C\setminus\{x\}}
            v(x\mid\{z\})
        \right)
    }{
        \displaystyle\sum_{y\in C}
        \exp\left(
            v(y)+\sum_{z\in C\setminus\{y\}}
            v(y\mid\{z\})
        \right)
    }.
    \label{eq:second-order-choice-probability}
\end{equation}
Here, $v(x)$ represents the intrinsic preference associated with $x$, while
$v(x\mid\{z\})$ represents the contribution of context alternative $z$ to
the utility of $x$.
\subsection{Reparameterisation of the Model}
The second-order formulation contains two types of parameters: the intrinsic
parameters $v(x)$ and the contextual parameters $v(x\mid\{z\})$. These can
be combined into a single set of pairwise parameters. For $x\neq z$, define
\begin{equation}
    u_{xz}=v(x\mid\{z\})-v(z).
    \label{eq:cdm-reparameterisation}
\end{equation}
Here, $x$ is the target alternative and $z$ is the context alternative. Using
this definition, the reparameterised utility is
\begin{align}
    \widetilde{u}(x\mid C)
    &=\sum_{z\in C\setminus\{x\}}u_{xz}\\
    &=\sum_{z\in C\setminus\{x\}}
      \left(v(x\mid\{z\})-v(z)\right)\\
    &=v(x)+\sum_{z\in C\setminus\{x\}}v(x\mid\{z\})
      -\sum_{z\in C}v(z)\\
    &=u(x\mid C)-\sum_{z\in C}v(z).
    \label{eq:reparameterised-utility}
\end{align}
The term $\sum_{z\in C}v(z)$ depends only on the choice set $C$ and is the
same for every candidate in that choice set. Subtracting the same quantity
from every candidate utility does not change the softmax probabilities:
\begin{equation}
    \frac{\exp(q_x-a)}{\sum_{y\in C}\exp(q_y-a)}
    =
    \frac{\exp(q_x)}{\sum_{y\in C}\exp(q_y)}.
\end{equation}
Therefore, reparameterisation changes the form of the parameters without
changing the choice probabilities.
\subsection{CDM Choice Probability}
After reparameterisation, the utility of alternative $x$ in choice set $C$ is
\begin{equation}
    \widetilde{u}(x\mid C)
    =
    \sum_{z\in C\setminus\{x\}}u_{xz}.
    \label{eq:cdm-utility}
\end{equation}
Therefore, the CDM choice probability is
\begin{equation}
    P(x\mid C)
    =
    \frac{
        \exp\left(
            \displaystyle\sum_{z\in C\setminus\{x\}}u_{xz}
        \right)
    }{
        \displaystyle\sum_{y\in C}
        \exp\left(
            \sum_{z\in C\setminus\{y\}}u_{yz}
        \right)
    }.
    \label{eq:full-rank-cdm-probability}
\end{equation}
\subsection{Interpretation of the Parameters}
The parameter $u_{xz}$ represents the effect of context alternative $z$ on
target alternative $x$. For example, $u_{\mathrm{Car},\mathrm{Bus}}$
represents the contextual effect of Bus on the utility associated with Car.
In general, $u_{xz}$ and $u_{zx}$ can be different because they represent
effects in opposite directions.
\subsection{Number of Parameters}
The reparameterised parameters can be arranged in a matrix
$U\in\R^{n\times n}$. The diagonal entries are set to zero because an
alternative is not treated as its own context:
\begin{equation}
    u_{xx}=0.
\end{equation}
Therefore, $U$ contains
\begin{equation}
    n^2-n=n(n-1)
\end{equation}
off-diagonal parameters. These parameters satisfy the identification
constraint
\begin{equation}
    \sum_{x\in\X}
    \sum_{z\in\X\setminus\{x\}}u_{xz}=0.
    \label{eq:cdm-identification-constraint}
\end{equation}
Hence, the number of free parameters is
\begin{equation}
    n(n-1)-1=n^2-n-1.
\end{equation}
This agrees with the count obtained from the original second-order
formulation in \cref{eq:second-order-parameter-count}.
For example, suppose $\X=\{A,B,C\}$. Then
\begin{equation}
    U=
    \begin{pmatrix}
        0      & u_{AB} & u_{AC}\\
        u_{BA} & 0      & u_{BC}\\
        u_{CA} & u_{CB} & 0
    \end{pmatrix}.
\end{equation}
The matrix has six off-diagonal parameters. After applying the constraint
\begin{equation}
    u_{AB}+u_{AC}+u_{BA}+u_{BC}+u_{CA}+u_{CB}=0,
\end{equation}
the number of free parameters is
\begin{equation}
    6-1=5.
\end{equation}
This is consistent with the five free parameters obtained in
\cref{eq:three-alternative-free-parameters}.

\chapter{Bradley--Terry Model and Maximum Likelihood}
\section{Pairwise Comparison Setting}
To understand the underlying mechanics of a model for choice data, we
consider observed pairwise comparisons and assume a probabilistic model.
The Bradley--Terry model is used for this purpose.
The goal is to estimate unknown parameters that provide a compressed
representation of the observed comparison data.
For items \(i\) and \(j\), with scores \(\theta_i\) and \(\theta_j\), the
Bradley--Terry model defines
\begin{equation}
    P(i\succ j)
    =
    \frac{\exp(\theta_i)}
    {\exp(\theta_i)+\exp(\theta_j)}.
    \label{eq:bradley-terry}
\end{equation}
Dividing the numerator and denominator by \(\exp(\theta_i)\) gives
\begin{equation}
    P(i\succ j)
    =
    \frac{1}
    {1+\exp\left(-(\theta_i-\theta_j)\right)}.
\end{equation}
Therefore, the probability depends only on the score difference
\(\theta_i-\theta_j\).
