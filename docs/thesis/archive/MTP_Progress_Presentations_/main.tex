\begin{frame}
  \frametitle{Low-rank CDM reparameterization}
  \begin{minipage}[t]{0.43\textwidth}
    \vspace{0pt}
    \sectionemphasis{Full CDM}
    \[
      u_i(S)=\sum_{\substack{j\in S\\j\neq i}} A_{ij},
      \qquad
      P(i\mid S)=\frac{e^{u_i(S)}}{\sum_{k\in S}e^{u_k(S)}}
    \]
    The interaction matrix (A\in\mathbb{R}^{n\times n}) contains one directed context effect for every item pair.
    \[
      \text{storage and estimation scale as } O(n^2)
    \]
  \end{minipage}
  \hfill
  \begin{minipage}[t]{0.50\textwidth}
    \vspace{0pt}
    \textcolor{IITMGold}{\textbf{Low-rank CDM}}
    \[
      A=TC^{\mathsf T},
      \qquad
      A_{ij}=t_i^{\mathsf T}c_j=\sum_{k=1}^{r}t_{ik}c_{jk}
    \]
    Item (i) receives a target vector (t_i\in\mathbb{R}^{r}). Item (j) receives a context vector (c_j\in\mathbb{R}^{r}).
    \[
      \text{storage and estimation scale as } O(nr)
    \]
  \end{minipage}
  \vfill
  \centering
  \result{Rank (r) controls how many latent interaction directions the model can represent.}
\end{frame}

\begin{frame}
  \frametitle{Rank selection and uniform performance}
  \begin{minipage}[t]{0.41\textwidth}
    \vspace{0pt}
    \sectionemphasis{Rank selection}
    \begin{enumerate}
      \item Fit several random initializations.
      \item Keep converged runs.
      \item Compare validation NLL.
      \item Evaluate the selected model once on test data.
    \end{enumerate}
    Validation controls model capacity without using the held-out test set.
  \end{minipage}
  \hfill
  \begin{minipage}[t]{0.53\textwidth}
    \vspace{0pt}
    \sectionemphasis{Uniform performance guarantee}
    \[
      \Pr\!\left(\sup_{\theta\in\Theta_r}\left|\widehat L_m(\theta)-L(\theta)\right|\leq\varepsilon\right)\geq 1-\delta
    \]
    With high probability, empirical loss stays close to population loss across the complete rank-(r) model class.

    \vspace{0.20cm}
    \result{More observations tighten the bound. Lower rank reduces model complexity.}

    \vspace{0.12cm}
    {\small Exact constants depend on the assumptions in Seshadri et al.}
  \end{minipage}
\end{frame}
