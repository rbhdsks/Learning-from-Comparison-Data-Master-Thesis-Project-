% 9. Reparameterization
\begin{frame}
  \frametitle{Low-rank CDM reparameterization}
  \small
  \begin{minipage}[t]{0.45\textwidth}
    \vspace{0pt}
    \sectionemphasis{Full CDM}
    \vspace{0.08cm}
    \[
      u_i(S)=\sum_{\substack{j\in S\\j\neq i}} A_{ij}
    \]
    \[
      P(i\mid S)=\frac{e^{u_i(S)}}{\sum_{k\in S}e^{u_k(S)}}
    \]
    $A\in\mathbb{R}^{n\times n}$ stores a directed context effect for each ordered item pair.

    \vspace{0.12cm}
    \centering\textbf{Parameter scale: $O(n^2)$}
  \end{minipage}
  \hfill
  \begin{minipage}[t]{0.45\textwidth}
    \vspace{0pt}
    \textcolor{IITMGold}{\textbf{Low-rank CDM}}
    \vspace{0.08cm}
    \[
      A=TC^{\mathsf T}
    \]
    \[
      A_{ij}=t_i^{\mathsf T}c_j
      =\sum_{k=1}^{r}t_{ik}c_{jk}
    \]
    Item $i$ has target vector $t_i\in\mathbb{R}^{r}$ and context vector $c_i\in\mathbb{R}^{r}$.

    \vspace{0.12cm}
    \centering\textbf{Parameter scale: $O(nr)$}
  \end{minipage}

  \vfill
  \centering
  {\footnotesize\result{Rank $r$ controls the number of latent interaction directions represented by the model.}}
\end{frame}

% 10. Rank selection and guarantee
\begin{frame}
  \frametitle{Rank selection and uniform performance guarantee}
  \small
  \begin{minipage}[t]{0.39\textwidth}
    \vspace{0pt}
    \sectionemphasis{Rank-selection procedure}
    \begin{enumerate}
      \item Fit each candidate rank using multiple initialization seeds.
      \item Retain converged solutions.
      \item Select rank using validation NLL.
      \item Evaluate the selected model once on test data.
    \end{enumerate}
    Validation controls capacity without reusing the held-out test set.
  \end{minipage}
  \hfill
  \begin{minipage}[t]{0.55\textwidth}
    \vspace{0pt}
    \sectionemphasis{Uniform performance guarantee}
    \[
      \Pr\!\left(
        \sup_{\theta\in\Theta_r}
        \left|\widehat L_m(\theta)-L(\theta)\right|
        \leq \varepsilon
      \right)\geq 1-\delta.
    \]
    With probability at least $1-\delta$, empirical loss stays within $\varepsilon$ of population loss for every model in the rank-$r$ class $\Theta_r$.

    \vspace{0.18cm}
    \result{More observations tighten the bound. Lower rank reduces model complexity.}

    \vspace{0.10cm}
    {\scriptsize Exact constants depend on the assumptions in Seshadri et al.}
  \end{minipage}
\end{frame}
