\begin{weeklybox}{What I did this week}
  \begin{itemize}
    \item Began studying Seshadri et al., \emph{Discovering Context Effects from Raw Choice Data}.
    \item Studied the Independence of Irrelevant Alternatives (IIA) property and its meaning in a choice model.
    \item Checked the odds-ratio condition
    \[
      \frac{P(x\mid A)}{P(y\mid A)}
      =
      \frac{P(x\mid B)}{P(y\mid B)}
    \]
    using a commute example with Car, Taxi, Bike, Bus, and Walking.
  \end{itemize}

\weekheading{2}{Earlier study phase}{Multinomial logit and parameter identification}

\begin{weeklybox}{What I did this week}
  \begin{itemize}
    \item Studied the multinomial logit probability
    \[
      P(x\mid C)=\frac{\exp(v(x))}{\sum_{y\in C}\exp(v(y))}.
    \]
    \item Verified that the relative odds of two alternatives do not depend on the other alternatives present in the choice set.
    \item Checked additive shift invariance numerically. Understood that a constraint such as $\sum_x v(x)=0$ is used for identification and is not an additional behavioural assumption.
  \end{itemize}
\end{weeklybox}

\weekheading{3}{Earlier study phase}{Universal logit and the decomposition of context effects}

\begin{weeklybox}{What I did this week}
  \begin{itemize}
    \item Studied how the universal logit model permits the utility $u(x\mid C)$ to depend on the complete choice set $C$.
    \item Worked through the decomposition
    \[
      u(x\mid C)=\sum_{B\subseteq C\setminus\{x\}}v(x\mid B)
    \]
    and the corresponding zero-sum constraints.
    \item Expanded the decomposition for the three-alternative set $\{C,T,Bi\}$ and obtained five free parameters after applying the constraints.
    \item Noted that the general parameter-count formula still needed to be checked carefully against the paper.
  \end{itemize}
\end{weeklybox}
